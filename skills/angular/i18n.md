# Angular Internationalization (i18n)

## 📋 Información

- **Skill ID**: `angular/i18n`
- **Versión**: 1.0.0
- **Categoría**: Angular
- **Prioridad**: Media
- **Angular Version**: 18+

## 🎯 Objetivo

Dominar la internacionalización en Angular para crear aplicaciones multiidioma accesibles globalmente.

---

## 📊 Enfoques de i18n

```
┌──────────────────────────────────────────────────────┐
│  1. COMPILE-TIME (Angular i18n)                      │
│     Genera builds separados por idioma               │
│     Mejor performance, más builds                    │
├──────────────────────────────────────────────────────┤
│  2. RUNTIME (Transloco/ngx-translate)                │
│     Cambia idioma sin reload                         │
│     Más flexible, pequeño overhead                   │
└──────────────────────────────────────────────────────┘
```

## 🎨 Angular i18n (Compile-Time)

### Setup Básico

```typescript
// angular.json
{
  "projects": {
    "my-app": {
      "i18n": {
        "sourceLocale": "en-US",
        "locales": {
          "es": {
            "translation": "src/locale/messages.es.xlf",
            "baseHref": "/es/"
          },
          "fr": {
            "translation": "src/locale/messages.fr.xlf",
            "baseHref": "/fr/"
          }
        }
      },
      "architect": {
        "build": {
          "configurations": {
            "es": {
              "localize": ["es"]
            },
            "fr": {
              "localize": ["fr"]
            }
          }
        }
      }
    }
  }
}
```

### Marcar Textos para Traducir

```typescript
@Component({
  selector: "app-welcome",
  template: `
    <!-- i18n básico -->
    <h1 i18n>Welcome to our app</h1>

    <!-- Con descripción y ID -->
    <p i18n="User greeting|@@userGreeting">Hello, {{ userName }}!</p>

    <!-- Atributos -->
    <img
      src="logo.png"
      i18n-alt="Image alt text|@@logoAlt"
      alt="Company logo"
    />

    <!-- Placeholders -->
    <p i18n>
      You have {{ count }}
      <ng-container i18n-plural [pluralMapping]="messageMapping">
        message
      </ng-container>
    </p>

    <!-- Select (gender, etc) -->
    <span i18n>
      {gender, select,
        male {Mr. {{ name }}}
        female {Ms. {{ name }}}
        other {{{ name }}}
      }
    </span>
  `,
})
export class WelcomeComponent {
  userName = "John";
  count = 5;
  gender = "male";
  name = "Smith";

  messageMapping: { [k: string]: string } = {
    "=0": "no messages",
    "=1": "one message",
    other: "# messages",
  };
}
```

### Pluralización

```typescript
@Component({
  template: `
    <p i18n>
      Updated
      {minutes, plural,
        =0 {just now}
        =1 {one minute ago}
        other {{{minutes}} minutes ago}
      }
    </p>

    <p i18n>
      {itemCount, plural,
        =0 {No items}
        =1 {One item}
        other {{{itemCount}} items}
      }
      in cart
    </p>
  `,
})
export class PluralComponent {
  minutes = 5;
  itemCount = 3;
}
```

### Extraer y Generar Traducciones

```bash
# Extraer textos traducibles
ng extract-i18n --output-path src/locale

# Genera: src/locale/messages.xlf

# Build con locale específico
ng build --configuration=es
ng build --configuration=fr

# Build todos los locales
ng build --localize
```

### Archivo de Traducción (XLIFF)

```xml
<!-- messages.es.xlf -->
<?xml version="1.0" encoding="UTF-8" ?>
<xliff version="1.2" xmlns="urn:oasis:names:tc:xliff:document:1.2">
  <file source-language="en-US" target-language="es" datatype="plaintext">
    <body>
      <trans-unit id="userGreeting" datatype="html">
        <source>Hello, <x id="INTERPOLATION"/>!</source>
        <target>¡Hola, <x id="INTERPOLATION"/>!</target>
        <context-group purpose="location">
          <context context-type="sourcefile">app/welcome.component.ts</context>
        </context-group>
        <note priority="1" from="description">User greeting</note>
      </trans-unit>

      <trans-unit id="8728351234567890" datatype="html">
        <source>Welcome to our app</source>
        <target>Bienvenido a nuestra aplicación</target>
      </trans-unit>
    </body>
  </file>
</xliff>
```

## 🌍 Runtime i18n (Transloco)

### Instalación

```bash
# Install Transloco (compatible with Angular 21)
npm install @jsverse/transloco@^7.0.0
ng g @jsverse/transloco:ng-add
```

### Configuración

```typescript
// transloco.config.ts
import { TranslocoGlobalConfig } from "@jsverse/transloco-utils";

const config: TranslocoGlobalConfig = {
  rootTranslationsPath: "src/assets/i18n/",
  langs: ["en", "es", "fr", "de"],
  keysManager: {},
};

export default config;
```

```typescript
// app.config.ts
import { provideTransloco } from "@jsverse/transloco";
import { TranslocoHttpLoader } from "./transloco-loader";

export const appConfig: ApplicationConfig = {
  providers: [
    provideTransloco({
      config: {
        availableLangs: ["en", "es", "fr", "de"],
        defaultLang: "en",
        reRenderOnLangChange: true,
        prodMode: !isDevMode(),
      },
      loader: TranslocoHttpLoader,
    }),
  ],
};
```

### Transloco HTTP Loader

```typescript
// transloco-loader.ts
import { inject, Injectable } from "@angular/core";
import { Translation, TranslocoLoader } from "@jsverse/transloco";
import { HttpClient } from "@angular/common/http";

@Injectable({ providedIn: "root" })
export class TranslocoHttpLoader implements TranslocoLoader {
  private http = inject(HttpClient);

  getTranslation(lang: string) {
    return this.http.get<Translation>(`/assets/i18n/${lang}.json`);
  }
}
```

### Uso en Templates

```typescript
import { TranslocoModule } from "@jsverse/transloco";

@Component({
  selector: "app-header",
  imports: [TranslocoModule],
  template: `
    <!-- Traducción simple -->
    <h1>{{ "welcome" | transloco }}</h1>

    <!-- Con parámetros -->
    <p>{{ "user.greeting" | transloco: { name: userName } }}</p>

    <!-- Directiva structural -->
    <ng-container *transloco="let t">
      <h2>{{ t("login.title") }}</h2>
      <input [placeholder]="t('login.username')" />
      <input [placeholder]="t('login.password')" type="password" />
      <button>{{ t("login.submit") }}</button>
    </ng-container>

    <!-- Con scope -->
    <div *transloco="let t; scope: 'admin'">
      {{ t("dashboard.title") }}
    </div>
  `,
})
export class HeaderComponent {
  userName = "John";
}
```

### Uso en Componentes

```typescript
import { TranslocoService } from "@jsverse/transloco";

@Component({
  selector: "app-user-profile",
  template: `
    <h1>{{ title }}</h1>
    <p>{{ message }}</p>
    <button (click)="showAlert()">Alert</button>
  `,
})
export class UserProfileComponent implements OnInit {
  private transloco = inject(TranslocoService);
  title = "";
  message = "";

  ngOnInit() {
    // Obtener traducción
    this.title = this.transloco.translate("user.profile.title");

    // Con parámetros
    this.message = this.transloco.translate("user.greeting", {
      name: "John",
    });

    // Observable
    this.transloco
      .selectTranslate("common.loading")
      .subscribe((text) => console.log(text));
  }

  showAlert() {
    const message = this.transloco.translate("user.profile.saved");
    alert(message);
  }
}
```

### Cambiar Idioma

```typescript
import { TranslocoService } from "@jsverse/transloco";

@Component({
  selector: "app-language-switcher",
  template: `
    <select
      [value]="currentLang"
      (change)="changeLang($any($event.target).value)"
    >
      @for (lang of availableLangs; track lang) {
        <option [value]="lang">{{ getLangName(lang) }}</option>
      }
    </select>

    <div class="flags">
      @for (lang of availableLangs; track lang) {
        <button
          (click)="changeLang(lang)"
          [class.active]="currentLang === lang"
        >
          {{ getLangFlag(lang) }}
        </button>
      }
    </div>
  `,
})
export class LanguageSwitcherComponent {
  private transloco = inject(TranslocoService);

  currentLang = this.transloco.getActiveLang();
  availableLangs = this.transloco.getAvailableLangs() as string[];

  changeLang(lang: string) {
    this.transloco.setActiveLang(lang);
    this.currentLang = lang;

    // Guardar preferencia
    localStorage.setItem("userLang", lang);
  }

  getLangName(lang: string): string {
    const names: Record<string, string> = {
      en: "English",
      es: "Español",
      fr: "Français",
      de: "Deutsch",
    };
    return names[lang] || lang;
  }

  getLangFlag(lang: string): string {
    const flags: Record<string, string> = {
      en: "🇬🇧",
      es: "🇪🇸",
      fr: "🇫🇷",
      de: "🇩🇪",
    };
    return flags[lang] || "🌍";
  }
}
```

### Detectar Idioma del Browser

```typescript
// app.config.ts
import { getBrowserLang } from "@jsverse/transloco";

export const appConfig: ApplicationConfig = {
  providers: [
    provideTransloco({
      config: {
        availableLangs: ["en", "es", "fr", "de"],
        defaultLang: getUserLang(),
        fallbackLang: "en",
        reRenderOnLangChange: true,
        prodMode: !isDevMode(),
      },
      loader: TranslocoHttpLoader,
    }),
  ],
};

function getUserLang(): string {
  // Primero checkear localStorage
  const savedLang = localStorage.getItem("userLang");
  if (savedLang) return savedLang;

  // Luego checkear browser
  const browserLang = getBrowserLang();
  const supportedLangs = ["en", "es", "fr", "de"];

  return supportedLangs.includes(browserLang!) ? browserLang! : "en";
}
```

## 🔤 Formateo de Fechas y Números

### Date Pipe con Locale

```typescript
import { DatePipe, registerLocaleData } from "@angular/common";
import localeEs from "@angular/common/locales/es";
import localeFr from "@angular/common/locales/fr";

// Registrar locales
registerLocaleData(localeEs);
registerLocaleData(localeFr);

@Component({
  selector: "app-date-demo",
  template: `
    <p>English: {{ today | date: "fullDate" : "" : "en-US" }}</p>
    <p>Spanish: {{ today | date: "fullDate" : "" : "es-ES" }}</p>
    <p>French: {{ today | date: "fullDate" : "" : "fr-FR" }}</p>

    <p>
      Currency: {{ price | currency: "EUR" : "symbol" : "1.2-2" : "es-ES" }}
    </p>
  `,
})
export class DateDemoComponent {
  today = new Date();
  price = 1234.56;
}
```

```typescript

import { TranslocoDatePipe } from "@jsverse/transloco-locale";
```

## 🌐 Testing con i18n

### Test con Transloco

```typescript
import { createComponentFactory, Spectator } from "@ngneat/spectator";
import { TranslocoTestingModule } from "@jsverse/transloco";

describe("MyComponent", () => {
  let spectator: Spectator<MyComponent>;

  const createComponent = createComponentFactory({
    component: MyComponent,
    imports: [
      TranslocoTestingModule.forRoot({
        langs: {
          en: {
            title: "Hello",
            message: "Welcome, {{name}}!",
          },
          es: {
            title: "Hola",
            message: "¡Bienvenido, {{name}}!",
          },
        },
        translocoConfig: {
          availableLangs: ["en", "es"],
          defaultLang: "en",
        },
      }),
    ],
  });

  beforeEach(() => (spectator = createComponent()));

  it("should display translated title", () => {
    expect(spectator.query("h1")).toHaveText("Hello");
  });

  it("should switch language", () => {
    spectator.component.transloco.setActiveLang("es");
    spectator.detectChanges();
    expect(spectator.query("h1")).toHaveText("Hola");
  });
});
```

## ⚠️ Anti-Patterns

### ❌ Hardcoded Text

```typescript
// ❌ MAL
template: `<h1>Welcome to my app</h1>`;

// ✅ BIEN
template: `<h1 i18n>Welcome to my app</h1>`;
// O con Transloco
template: `<h1>{{ 'welcome' | transloco }}</h1>`;
```

### ❌ Concatenar Strings Traducidos

```typescript
// ❌ MAL
const message = this.t("hello") + " " + userName + "!";

// ✅ BIEN: Usar interpolación
const message = this.t("user.greeting", { name: userName });
// JSON: "user.greeting": "Hello, {{name}}!"
```

### ❌ No Manejar Plurales

```typescript
// ❌ MAL
template: `{{ count }} item(s)`;

// ✅ BIEN
template: `
  {count, plural,
    =0 {No items}
    =1 {One item}
    other {{{count}} items}
  }
`;
```
